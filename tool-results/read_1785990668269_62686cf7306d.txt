    31→  { label: "Karachi Coast", seed: "karachi" },
    32→  { label: "Hunza Valley", seed: "hunza" },

    37→export default function PakistanCraft() {
    38→  const canvasRef = useRef<HTMLCanvasElement>(null);
    39→  const engineRef = useRef<Engine | null>(null);
    40→  const seedRef = useRef<number>(0);
    41→  const [phase, setPhase] = useState<Phase>("start");
    42→  const [hud, setHud] = useState<HudState | null>(null);
    43→  const [seedInput, setSeedInput] = useState("");
    44→  const [showInventory, setShowInventory] = useState(false);
    45→  const [paused, setPaused] = useState(false);
    46→  const [showSettings, setShowSettings] = useState(false);
    47→  const [error, setError] = useState<string | null>(null);
    48→  const [hasSave, setHasSave] = useState(() =>
    49→    typeof window !== "undefined" ? Engine.hasSave() : false
    50→  );
    51→  const [settings, setSettings] = useState<Settings>(() =>
    52→    typeof window !== "undefined" ? loadSettings() : DEFAULT_SETTINGS
    53→  );
    54→  const [toast, setToast] = useState<string | null>(null);
    55→  const [engineState, setEngineState] = useState<Engine | null>(null);
    56→  const [forceTouch, setForceTouch] = useState(false);
    57→  const detectedTouch = useIsTouchDevice();
    58→  const isTouch = forceTouch || detectedTouch;
    59→
    60→  // refresh save flag when returning to start screen
    61→  useEffect(() => {
    62→    if (phase === "start") {
    63→      const flag = Engine.hasSave();
    64→      queueMicrotask(() => setHasSave(flag));
    65→    }
    66→  }, [phase]);
    67→
    68→  const showToast = useCallback((msg: string) => {
    69→    setToast(msg);
    70→    setTimeout(() => setToast(null), 2200);
    71→  }, []);
    72→
    73→  const startGame = useCallback(
    74→    (seedStr: string, loadSave: boolean) => {
    75→      const seedNum = seedStr ? hashSeed(seedStr) : Math.floor(Math.random() * 1e9);
    76→      seedRef.current = seedNum;
    77→      setHud(null);
    78→      setError(null);
    79→      setPaused(false);
    80→      setShowSettings(false);
    81→      setShowInventory(false);
    82→      setPhase("playing");
    83→      // stash whether to load save (handled in mount effect)
    84→      if (loadSave) {
    85→        sessionStorage.setItem("pc.loadsave", "1");
    86→      } else {
    87→        sessionStorage.removeItem("pc.loadsave");
    88→      }
    89→    },
    90→    []
    91→  );
    92→
    93→  // mount engine when entering playing phase
    94→  useEffect(() => {
    95→    if (phase !== "playing" || !canvasRef.current) return;
    96→    const s = loadSettings();
    97→    queueMicrotask(() => setSettings(s));
    98→    const wantLoad = sessionStorage.getItem("pc.loadsave") === "1";
    99→    sessionStorage.removeItem("pc.loadsave");
   100→    let eng: Engine | null = null;
   101→    try {
   102→      eng = new Engine(canvasRef.current, {
   103→        seed: seedRef.current,
   104→        settings: s,
   105→        onHud: (st) => setHud(st),
   106→        onError: (m) => setError(m),
   107→      });
   108→    } catch (e) {
   109→      const msg =
   110→        "Failed to start the game engine. " +
   111→        (e instanceof Error ? e.message : String(e));
   112→      queueMicrotask(() => setError(msg));
   113→      return;
   114→    }
   115→    engineRef.current = eng;
   116→    setEngineState(eng);
   117→    const raf2 = requestAnimationFrame(() => {
   118→      requestAnimationFrame(() => {
   119→        try {
   120→          eng?.start();
   121→          if (wantLoad && Engine.hasSave()) {
   122→            const r = eng.loadState();
   123→            if (r) {
   124→              seedRef.current = r.seed;
   125→              showToast("World loaded");
   126→            }
   127→          }
   128→        } catch (e) {
   129→          const msg =
   130→            "Failed to generate world. " +
   131→            (e instanceof Error ? e.message : String(e));
   132→          queueMicrotask(() => setError(msg));
   133→        }
   134→      });
   135→    });
   136→    return () => {
   137→      cancelAnimationFrame(raf2);
   138→      eng?.dispose();
   139→      engineRef.current = null;
   140→      setEngineState(null);
   141→    };
   142→  }, [phase, showToast]);
   143→
   144→  // keyboard: E inventory, Esc pause/settings
   145→  useEffect(() => {
   146→    if (phase !== "playing") return;
   147→    const handler = (e: KeyboardEvent) => {
   148→      if (e.code === "KeyE") {
   149→        e.preventDefault();
   150→        if (showSettings || paused) return;
   151→        setShowInventory((v) => {
   152→          const next = !v;
   153→          const eng = engineRef.current;
   154→          if (eng) {
   155→            eng.inputEnabled = !next;
   156→            if (next && document.pointerLockElement) document.exitPointerLock();
   157→          }
   158→          return next;
   159→        });
   160→      } else if (e.code === "Escape") {
   161→        if (showInventory) {
   162→          setShowInventory(false);
   163→          const eng = engineRef.current;
   164→          if (eng) eng.inputEnabled = true;
   165→        } else if (showSettings) {
   166→          setShowSettings(false);
   167→        } else {
   168→          setPaused((p) => {
   169→            const next = !p;
   170→            const eng = engineRef.current;
   171→            if (eng) eng.inputEnabled = !next;
   172→            if (next && document.pointerLockElement) document.exitPointerLock();
   173→            return next;
   174→          });
   175→        }
   176→      }
   177→    };
   178→    window.addEventListener("keydown", handler);
   179→    return () => window.removeEventListener("keydown", handler);
   180→  }, [phase, showInventory, paused, showSettings]);
   181→
   182→  const closeInventory = useCallback(() => {
   183→    setShowInventory(false);
   184→    const eng = engineRef.current;
   185→    if (eng) {
   186→      eng.inputEnabled = true;
   187→      eng.requestPointerLock();
   188→    }
   189→  }, []);
   190→
   191→  const resume = useCallback(() => {
   192→    setPaused(false);
   193→    const eng = engineRef.current;
   194→    if (eng) {
   195→      eng.inputEnabled = true;
   196→      eng.requestPointerLock();
   197→    }
   198→  }, []);
   199→
   200→  const openSettings = useCallback(() => setShowSettings(true), []);
   201→  const closeSettings = useCallback(() => setShowSettings(false), []);
   202→
   203→  const onSaveGame = useCallback(() => {
   204→    const eng = engineRef.current;
   205→    if (!eng) return;
   206→    const ok = eng.saveState(seedRef.current);
   207→    if (ok) {
   208→      setHasSave(true);
   209→      showToast("Game saved");
   210→    } else {
   211→      showToast("Save failed");
   212→    }
   213→  }, [showToast]);
   214→
   215→  const onQuitToMenu = useCallback(() => {
   216→    const eng = engineRef.current;
   217→    eng?.dispose();
   218→    engineRef.current = null;
   219→    setPhase("start");
   220→    setPaused(false);
   221→    setShowSettings(false);
   222→    setShowInventory(false);
   223→    setHud(null);
   224→    setHasSave(Engine.hasSave());
   225→  }, []);
   226→
   227→  const onDeleteSave = useCallback(() => {
   228→    Engine.clearSave();
   229→    setHasSave(false);
   230→    showToast("Save deleted");
   231→  }, [showToast]);
   232→
   233→  const onCanvasClick = useCallback(() => {
   234→    if (showInventory || paused || showSettings) return;
   235→    engineRef.current?.requestPointerLock();
   236→  }, [showInventory, paused, showSettings]);
   237→
   238→  const pickBlock = useCallback(
   239→    (blockId: number) => {
   240→      const eng = engineRef.current;
   241→      if (!eng || !hud) return;
   242→      eng.setHotbarSlot(eng.selected, blockId);
   243→    },
   244→    [hud]
   245→  );
   246→
   247→  const updateSettings = useCallback((partial: Partial<Settings>) => {
   248→    const eng = engineRef.current;
   249→    if (eng) {
   250→      eng.setSettings(partial);
   251→      setSettings({ ...eng.settings });
   252→    } else {
   253→      const next = { ...loadSettings(), ...partial };
   254→      saveSettings(next);
   255→      setSettings(next);
   256→    }
   257→  }, []);
   258→
   259→  if (error) {
   260→    return <ErrorScreen message={error} onBack={() => setPhase("start")} />;
   261→  }
   262→
   263→  return (
   264→    <div
   265→      className="fixed inset-0 overflow-hidden bg-black select-none"
   266→      style={{ height: "100dvh" }}
   267→    >
   268→      {phase === "playing" && (
   269→        <canvas
   270→          ref={canvasRef}
   271→          onClick={onCanvasClick}
   272→          className="absolute inset-0 block h-full w-full"
   273→          style={{ touchAction: "none" }}
   274→        />
   275→      )}
   276→
   277→      {phase === "start" && (
   278→        <StartScreen
   279→          onPlay={(s) => startGame(s, false)}
   280→          onContinue={() => startGame("", true)}
   281→          hasSave={hasSave}
   282→          onDeleteSave={onDeleteSave}
   283→          settings={settings}
   284→          onUpdateSettings={updateSettings}
   285→        />
   286→      )}
   287→
   288→      {phase === "playing" && !hud && <LoadingScreen />}
   289→
   290→      {phase === "playing" && hud && (
   291→        <>
   292→          <Hud hud={hud} isTouch={isTouch} />
   293→          <Minimap hud={hud} engine={engineState} />
   294→          {isTouch && !paused && !showSettings && !showInventory && (
   295→            <TouchControls
   296→              onMove={(x, y) => engineRef.current?.setTouchInput({ moveX: x, moveY: y })}
   297→              onLook={(dx, dy) => engineRef.current?.setTouchInput({ lookDX: dx, lookDY: dy })}
   298→              onJump={(d) => engineRef.current?.setTouchInput({ jump: d })}
   299→              onSprint={(d) => engineRef.current?.setTouchInput({ sprint: d })}
   300→              onSneak={(d) => engineRef.current?.setTouchInput({ sneak: d })}
   301→              onBreakStart={() => engineRef.current?.setTouchInput({ breakBtn: true })}
   302→              onBreakEnd={() => engineRef.current?.setTouchInput({ breakBtn: false })}
   303→              onTapBreak={() => engineRef.current?.breakBlock()}
   304→              onPlace={() => engineRef.current?.placeBlock()}
   305→              onFly={() => {
   306→                const eng = engineRef.current;
   307→                if (!eng) return;
   308→                if (eng.player.gameMode !== "creative") eng.setGameMode("creative");
   309→                eng.player.toggleFly();
   310→              }}
   311→              onInventory={() => {
   312→                setShowInventory(true);
   313→                const eng = engineRef.current;
   314→                if (eng) {
   315→                  eng.inputEnabled = false;
   316→                  if (document.pointerLockElement) document.exitPointerLock();
   317→                }
   318→              }}
   319→              hotbar={hud.hotbar}
   320→              selectedSlot={hud.selectedSlot}
   321→              onSelectSlot={(i) => engineRef.current?.setSelected(i)}
   322→            />
   323→          )}
   324→          {paused && !showSettings && (
   325→            <PauseOverlay
   326→              onResume={resume}
   327→              onSettings={openSettings}
   328→              onSave={onSaveGame}
   329→              onQuit={onQuitToMenu}
   330→              hasSave={hasSave}
   331→              isTouch={isTouch}
   332→              onToggleTouch={() => setForceTouch((v) => !v)}
   333→            />
   334→          )}
   335→          {showSettings && (
   336→            <SettingsPanel
   337→              settings={settings}
   338→              onUpdate={updateSettings}
   339→              onClose={closeSettings}
   340→              inGame
   341→            />
   342→          )}
   343→          {showInventory && (
   344→            <Inventory
   345→              selectedSlot={hud.selectedSlot}
   346→              hotbar={hud.hotbar}
   347→              onPick={pickBlock}
   348→              onClose={closeInventory}
   349→            />
   350→          )}
   351→        </>
   352→      )}
   353→
   354→      {toast && (
   355→        <div className="pointer-events-none absolute bottom-28 left-1/2 z-50 -translate-x-1/2 rounded-lg border border-white/15 bg-black/80 px-4 py-2 font-mono text-sm text-emerald-300 backdrop-blur">
   356→          {toast}
   357→        </div>
   358→      )}
   359→    </div>
   360→  );
   361→}
