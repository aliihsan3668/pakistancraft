   139→// Custom water shader: animated vertex waves + specular shimmer + flowing texture,
   140→// respects vertex color (AO/directional shade) and fog.
   141→function makeWaterMaterial(atlas: THREE.Texture): THREE.ShaderMaterial {
   142→  return new THREE.ShaderMaterial({
   143→    transparent: true,
   144→    depthWrite: false,
   145→    side: THREE.DoubleSide,
   146→    fog: false, // we handle fog manually in the shader
   147→    uniforms: {
   148→      uAtlas: { value: atlas },
   149→      uTime: { value: 0 },
   150→      uFogColor: { value: new THREE.Color("#bcd8ee") },
   151→      uFogNear: { value: 32 },
   152→      uFogFar: { value: 72 },
   153→      uOpacity: { value: 0.82 },
   154→      uSunDir: { value: new THREE.Vector3(0, 1, 0) },
   155→      uSunColor: { value: new THREE.Color("#fff0c0") },
   156→    },
   157→    vertexShader: /* glsl */ `
   158→      attribute vec3 color;
   159→      varying vec2 vUv;
   160→      varying vec3 vColor;
   161→      varying vec3 vWorldPos;
   162→      varying float vFogDepth;
   163→      varying float vIsTop;
   164→      uniform float uTime;
   165→      void main() {
   166→        vUv = uv;
   167→        vColor = color;
   168→        vec3 pos = position;
   169→        vIsTop = normal.y > 0.5 ? 1.0 : 0.0;
   170→        // wave displacement on top faces
   171→        if (vIsTop > 0.5) {
   172→          float w = sin(pos.x * 0.8 + uTime * 1.5) * 0.05
   173→                  + cos(pos.z * 0.6 + uTime * 1.2) * 0.05
   174→                  + sin((pos.x + pos.z) * 0.4 + uTime * 0.8) * 0.03;
   175→          pos.y += w;
   176→        }
   177→        vec4 worldPos = modelMatrix * vec4(pos, 1.0);
   178→        vWorldPos = worldPos.xyz;
   179→        vec4 mv = modelViewMatrix * vec4(pos, 1.0);
   180→        vFogDepth = -mv.z;
   181→        gl_Position = projectionMatrix * mv;
   182→      }
   183→    `,
   184→    fragmentShader: /* glsl */ `
   185→      uniform sampler2D uAtlas;
   186→      uniform float uTime;
   187→      uniform vec3 uFogColor;
   188→      uniform float uFogNear;
   189→      uniform float uFogFar;
   190→      uniform float uOpacity;
   191→      uniform vec3 uSunDir;
   192→      uniform vec3 uSunColor;
   193→      varying vec2 vUv;
   194→      varying vec3 vColor;
   195→      varying vec3 vWorldPos;
   196→      varying float vFogDepth;
   197→      varying float vIsTop;
   198→      void main() {
   199→        // scroll UVs for a flowing feel
   200→        vec2 uv = vUv + vec2(uTime * 0.012, uTime * 0.018);
   201→        vec4 tex = texture2D(uAtlas, uv);
   202→        vec3 col = tex.rgb * vColor;
   203→        // specular shimmer on top faces (sun glint)
   204→        if (vIsTop > 0.5) {
   205→          // approximate normal from wave derivatives
   206→          vec3 n = normalize(vec3(
   207→            cos(vWorldPos.x * 0.8 + uTime * 1.5) * 0.3,
   208→            1.0,
   209→            sin(vWorldPos.z * 0.6 + uTime * 1.2) * 0.3
   210→          ));
   211→          vec3 viewDir = normalize(cameraPosition - vWorldPos);
   212→          vec3 halfDir = normalize(normalize(uSunDir) + viewDir);
   213→          float spec = pow(max(dot(n, halfDir), 0.0), 32.0);
   214→          col += uSunColor * spec * 0.6;
   215→          // slight depth-based color shift (deeper = darker)
   216→          col *= 0.85 + 0.15 * smoothstep(0.0, 0.5, vColor.r);
   217→        }
   218→        // fog
   219→        float fogFactor = smoothstep(uFogNear, uFogFar, vFogDepth);
   220→        col = mix(col, uFogColor, fogFactor);
   221→        float alpha = uOpacity * (1.0 - fogFactor * 0.2);
   222→        gl_FragColor = vec4(col, alpha);
   223→      }
   224→    `,
   225→    vertexColors: true,
   226→  });
   227→}

   229→function headingFromYaw(yaw: number): string {
   230→  // yaw=0 → looking -Z (north). Normalize to 0..2π.
   231→  let a = (-yaw) % (Math.PI * 2);
   232→  if (a < 0) a += Math.PI * 2;
   233→  const deg = (a / (Math.PI * 2)) * 360;
   234→  const dirs = ["N", "NE", "E", "SE", "S", "SW", "W", "NW"];
   235→  const idx = Math.round(deg / 45) % 8;
   236→  return dirs[idx];
   237→}

   486→    this.running = false;

   488→    window.removeEventListener("keydown", this._onKeyDown);
   489→    window.removeEventListener("keyup", this._onKeyUp);
   490→    window.removeEventListener("mousedown", this._onMouseDown);
   491→    window.removeEventListener("mouseup", this._onMouseUp);
   492→    window.removeEventListener("mousemove", this._onMouseMove);
   493→    window.removeEventListener("wheel", this._onWheel);
   494→    document.removeEventListener("pointerlockchange", this._onPointerLockChange);
   495→    window.removeEventListener("resize", this._onResize);
   496→    if (typeof window !== "undefined" && window.visualViewport) {
   497→      window.visualViewport.removeEventListener("resize", this._onResize);
   498→    }
   499→    this.canvas.removeEventListener("contextmenu", this._onContext);
   500→    if (document.pointerLockElement === this.canvas) document.exitPointerLock();
   501→    // dispose all subsystems to prevent GPU memory leaks
   502→    this.world.dispose();
   503→    this.sky.dispose();
   504→    this.particles.dispose();
   505→    if (this.clouds) {
   506→      this.scene.remove(this.clouds);
   507→      this.clouds.geometry.dispose();
   508→      (this.clouds.material as THREE.Material).dispose();
   509→    }
   510→    if (this.rain) {
   511→      this.scene.remove(this.rain);
   512→      this.rain.geometry.dispose();
   513→      (this.rain.material as THREE.Material).dispose();
   514→    }
   515→    if (this.selectionBox) {
   516→      this.scene.remove(this.selectionBox);
   517→      this.selectionBox.geometry.dispose();
   518→      (this.selectionBox.material as THREE.Material).dispose();
   519→    }
   520→    this.materials.solid.dispose();
   521→    this.materials.cutout.dispose();
   522→    this.materials.water.dispose();
   523→    this.renderer.dispose();
   524→  }

   526→  private initSpawn() {
   527→    for (let dx = -2; dx <= 2; dx++)
   528→      for (let dz = -2; dz <= 2; dz++) this.world.ensureTerrain(dx, dz);
   529→    for (let dx = -2; dx <= 2; dx++)
   530→      for (let dz = -2; dz <= 2; dz++) this.world.ensureDecorated(dx, dz);
   531→    // Spawn the player at the Lahore city center, elevated for a panoramic view.
   532→    const spawnX = 40;
   533→    const spawnZ = 40;
   534→    const groundY = Player.spawnY(this.world, spawnX, spawnZ);
   535→    // Spawn on the ground (not flying) to avoid constant chunk streaming.
   536→    this.player.pos.set(spawnX + 0.5, groundY + 0.5, spawnZ + 0.5);
   537→    this.player.vel.set(0, 0, 0);
   538→    this.player.yaw = 0.7; // face toward the mosque entrance
   539→    this.player.pitch = -0.1;
   540→    (this.player as unknown as { _targetYaw: number })._targetYaw = 0.7;
   541→    (this.player as unknown as { _targetPitch: number })._targetPitch = -0.1;
   542→    // creative mode but not flying — press F to fly
   543→    this.player.gameMode = "creative";
   544→    this.player.flying = false;
   545→  }

   733→    const code = e.code;

   735→    if (code.startsWith("Digit")) {

   740→      // fly toggle — creative only
   741→      if (this.player.gameMode === "creative") this.player.toggleFly();

   743→    if (code === "KeyG") {
   744→      // toggle game mode
   745→      this.setGameMode(

   750→      // rain toggle

   752→      this.weatherTimer = 120;

   755→      this.setTimeOfDay(0.5); // noon
   756→    }
   757→    if (code === "KeyN") {
   758→      this.setTimeOfDay(0.0); // night
   759→    }
   760→    if (code === "KeyH") {
   761→      // teleport to Lahore city center
   762→      this.teleportTo(40, 40);
   763→    }
   764→    if (code === "Escape") {
   765→      if (document.pointerLockElement === this.canvas) document.exitPointerLock();
   766→    }
   767→    if (code === "Space" || code.startsWith("Arrow")) e.preventDefault();

   803→  private onMouseDown(e: MouseEvent) {
   804→    this.mouseButtons.add(e.button);
   805→    if (!this.inputEnabled) return;
   806→    if (this.pointerLocked) {
   807→      if (e.button === 0) {
   808→        this.tryBreak();
   809→        this.breakCooldown = 0.25;
   810→      } else if (e.button === 2) {
   811→        this.tryPlace();
   812→        this.placeCooldown = 0.25;
   813→      }
   814→    } else {
   815→      if (e.button === 0) {
   816→        this.dragLook = true;
   817→        this.dragLastX = e.clientX;
   818→        this.dragLastY = e.clientY;
   819→        this.clickStartX = e.clientX;
   820→        this.clickStartY = e.clientY;
   821→        this.clickStartT = performance.now();
   822→        this.clickMoved = 0;
   823→      } else if (e.button === 2) {
   824→        this.tryPlace();
   825→        this.placeCooldown = 0.25;
   826→      }
   827→    }
   828→  }
   829→
   830→  private onMouseUp(e: MouseEvent) {
   831→    this.mouseButtons.delete(e.button);
   832→    if (!this.inputEnabled) {
   833→      this.dragLook = false;
   834→      return;
   835→    }
   836→    if (!this.pointerLocked && e.button === 0 && this.dragLook) {
   837→      const dt = performance.now() - this.clickStartT;
   838→      if (this.clickMoved < 6 && dt < 300) {
   839→        this.tryBreak();
   840→        this.breakCooldown = 0.25;
   841→      }
   842→      this.dragLook = false;
   843→    }
   844→  }

   846→  private onMouseMove(e: MouseEvent) {
   847→    if (!this.inputEnabled) return;
   848→    const sens = 0.0022 * this.settings.mouseSensitivity;
   849→    if (this.pointerLocked) {
   850→      this.player.setLookFromDelta(e.movementX, e.movementY, sens);
   851→    } else if (this.dragLook && this.mouseButtons.has(0)) {
   852→      const dx = e.clientX - this.dragLastX;
   853→      const dy = e.clientY - this.dragLastY;
   854→      this.dragLastX = e.clientX;
   855→      this.dragLastY = e.clientY;
   856→      this.clickMoved += Math.abs(dx) + Math.abs(dy);
   857→      this.player.setLookFromDelta(dx, dy, 0.005 * this.settings.mouseSensitivity);
   858→    }
   859→  }

   892→  private tryBreak() {
   893→    const hit = this.player.raycast(REACH_DISTANCE);
   894→    if (!hit) return;
   895→    const b = this.world.getBlock(hit.x, hit.y, hit.z);
   896→    if (b === Block.BEDROCK) return;
   897→    this.world.setBlock(hit.x, hit.y, hit.z, Block.AIR);
   898→    this.edits.set(`${hit.x},${hit.y},${hit.z}`, Block.AIR);
   899→    this.particles.emitBlockBreak(hit.x, hit.y, hit.z, b, 16);
   900→    this.world.remeshAround(hit.x, hit.z);
   901→    if (hit.nx !== 0) this.world.remeshAround(hit.x + hit.nx, hit.z);
   902→    if (hit.nz !== 0) this.world.remeshAround(hit.x, hit.z + hit.nz);
   903→  }

   905→  private tryPlace() {
   906→    const hit = this.player.raycast(REACH_DISTANCE);
   907→    if (!hit) return;
   908→    // If looking at TNT, ignite it instead of placing
   909→    const lookedAt = this.world.getBlock(hit.x, hit.y, hit.z);
   910→    if (lookedAt === Block.TNT) {
   911→      this.explosions.ignite(hit.x, hit.y, hit.z);
   912→      return;
   913→    }
   914→    const px = hit.x + hit.nx;
   915→    const py = hit.y + hit.ny;
   916→    const pz = hit.z + hit.nz;
   917→    const playerMinX = this.player.pos.x - 0.3;
   918→    const playerMaxX = this.player.pos.x + 0.3;
   919→    const playerMinY = this.player.pos.y;
   920→    const playerMaxY = this.player.pos.y + 1.8;
   921→    const playerMinZ = this.player.pos.z - 0.3;
   922→    const playerMaxZ = this.player.pos.z + 0.3;
   923→    if (
   924→      px + 1 > playerMinX &&
   925→      px < playerMaxX &&
   926→      py + 1 > playerMinY &&
   927→      py < playerMaxY &&
   928→      pz + 1 > playerMinZ &&
   929→      pz < playerMaxZ
   930→    ) {
   931→      const def = BLOCKS[this.getSelectedBlock()];
   932→      if (def && def.solid) return;
   933→    }
   934→    const existing = this.world.getBlock(px, py, pz);
   935→    if (existing !== Block.AIR && !BLOCKS[existing]?.liquid) return;
   936→    const placed = this.getSelectedBlock();
   937→    this.world.setBlock(px, py, pz, placed);
   938→    this.edits.set(`${px},${py},${pz}`, placed);
   939→    this.particles.emitBlockBreak(px, py, pz, placed, 6);
   940→    this.world.remeshAround(px, pz);
   941→    if (hit.nx !== 0) this.world.remeshAround(px + hit.nx, pz);
   942→    if (hit.nz !== 0) this.world.remeshAround(px, pz + hit.nz);
   943→  }

   949→    canvas.width = size;
   950→    canvas.height = size;

   959→      g.addColorStop(1, "rgba(255,255,255,0)");

   973→      opacity: 0.6,

  1095→    );
  1096→
  1097→    const ang = (this.timeOfDay - 0.25) * Math.PI * 2;
  1098→    const sunY = Math.sin(ang);
  1099→    const sunX = Math.cos(ang);
  1100→    this.sun.position.set(
  1101→      this.player.pos.x + sunX * 120,
  1102→      this.player.pos.y + sunY * 120 + 10,
  1103→      this.player.pos.z + 40
  1104→    );
  1105→    this.sun.target.position.copy(this.player.pos);
  1106→
  1107→    const dayFactor = Math.max(0, sunY);
  1108→    let sunI = 0.2 + dayFactor * 1.0;
  1109→    let ambI = 0.18 + dayFactor * 0.3;
  1110→    let hemiI = 0.25 + dayFactor * 0.45;
  1111→
  1112→    // weather dimming
  1113→    if (this.raining) {
  1114→      sunI *= 0.45;
  1115→      ambI *= 0.7;
  1116→      hemiI *= 0.6;
  1117→    }
  1118→
  1119→    const warmth = Math.max(0, 1 - Math.abs(sunY) * 2.2);
  1120→    const sunColor = this._sunColor.lerpColors(this.C_SUN_WARM, this.C_SUN_SUNSET, warmth);
  1121→    if (this.raining) sunColor.multiplyScalar(0.7);
  1122→    this.sun.color.copy(sunColor);
  1123→    this.sun.intensity = sunI * this.settings.brightness;
  1124→    this.ambient.intensity = ambI * this.settings.brightness;
  1125→    this.hemi.intensity = hemiI * this.settings.brightness;
  1126→
  1127→    // background + fog follow sky horizon color
  1128→    (this.scene.background as THREE.Color).copy(this.sky.skyColor);
  1129→    (this.scene.fog as THREE.Fog).color.copy(this.sky.fogColor);
  1130→    if (this.raining) {
  1131→      (this.scene.background as THREE.Color).multiplyScalar(0.6);
  1132→      (this.scene.fog as THREE.Fog).color.multiplyScalar(0.7);
  1133→    }
  1134→
  1135→    // clouds follow player
  1136→    if (this.clouds) {
  1137→      this.clouds.position.x = this.player.pos.x;
  1138→      this.clouds.position.z = this.player.pos.z;
  1139→      const cm = this.clouds.material as THREE.MeshBasicMaterial;
  1140→      let cloudOp = 0.3 + dayFactor * 0.4;
  1141→      if (this.raining) cloudOp = 0.7;
  1142→      cm.opacity = cloudOp;
  1143→    }
  1144→
  1145→    // water shader uniforms
  1146→    const wmat = this.materials.water as THREE.ShaderMaterial;
  1147→    wmat.uniforms.uTime.value += dt;
  1148→    wmat.uniforms.uFogColor.value.copy(this.sky.fogColor);
  1149→    wmat.uniforms.uFogNear.value = CHUNK_SIZE * 2;
  1150→    wmat.uniforms.uFogFar.value = CHUNK_SIZE * (this.settings.renderDistance - 0.5);
  1151→    // sun direction for specular shimmer
  1152→    const wAng = (this.timeOfDay - 0.25) * Math.PI * 2;
  1153→    (wmat.uniforms.uSunDir.value as THREE.Vector3).set(
  1154→      Math.cos(wAng), Math.sin(wAng), 0.3
  1155→    ).normalize();
  1156→    (wmat.uniforms.uSunColor.value as THREE.Color).copy(this.sun.color);
  1157→  }
  1158→
  1159→  // ---- main loop ----
  1160→  private loop = (now: number) => {
  1161→    if (!this.running) return;
  1162→    this._raf = requestAnimationFrame(this.loop);
  1163→    // Cap dt at 33ms (30fps min) to prevent physics explosions on frame spikes.
  1164→    // If the frame took longer, we run multiple sub-steps for stable physics.
  1165→    const rawDt = (now - this.lastTime) / 1000;
  1166→    this.lastTime = now;
  1167→    const dt = Math.min(0.033, rawDt);
  1168→
  1169→    // fps
  1170→    this.fpsAccum += dt;
  1171→    this.fpsCount++;
  1172→    if (this.fpsAccum >= 0.5) {
  1173→      this.fps = Math.round(this.fpsCount / this.fpsAccum);
  1174→      this.fpsAccum = 0;
  1175→      this.fpsCount = 0;
  1176→    }
  1177→
  1178→    // input
  1179→    this.updateInputState();
  1180→
  1181→    // look: touch + arrow keys + mouse (mouse handled in onMouseMove)
  1182→    if (this.inputEnabled) {
  1183→      const lr = 1.4 * dt;
  1184→      if (this.keys.has("ArrowLeft"))
  1185→        this.player.setLookFromDelta(-lr, 0, 1);
  1186→      if (this.keys.has("ArrowRight"))
  1187→        this.player.setLookFromDelta(lr, 0, 1);
  1188→      if (this.keys.has("ArrowUp"))
  1189→        this.player.setLookFromDelta(0, -lr, 1);
  1190→      if (this.keys.has("ArrowDown"))
  1191→        this.player.setLookFromDelta(0, lr, 1);
  1192→      // touch look — higher sensitivity than mouse for responsive feel
  1193→      if (this.touchInput.lookDX !== 0 || this.touchInput.lookDY !== 0) {
  1194→        this.player.setLookFromDelta(
  1195→          this.touchInput.lookDX,
  1196→          this.touchInput.lookDY,
  1197→          0.006 * this.settings.mouseSensitivity
  1198→        );
  1199→        this.touchInput.lookDX = 0;
  1200→        this.touchInput.lookDY = 0;
  1201→      }
  1202→    }
  1203→
  1204→    // continuous break/place (mouse + touch buttons)
  1205→    if (this.inputEnabled) {
  1206→      this.breakCooldown -= dt;
  1207→      this.placeCooldown -= dt;
  1208→      if (
  1209→        this.pointerLocked &&
  1210→        this.mouseButtons.has(0) &&
  1211→        this.breakCooldown <= 0
  1212→      ) {
  1213→        this.tryBreak();
  1214→        this.breakCooldown = 0.22;
  1215→      }
  1216→      if (this.mouseButtons.has(2) && this.placeCooldown <= 0) {
  1217→        this.tryPlace();
  1218→        this.placeCooldown = 0.22;
  1219→      }
  1220→      // touch break button
  1221→      if (this.touchInput.breakBtn && this.breakCooldown <= 0) {
  1222→        this.tryBreak();
  1223→        this.breakCooldown = 0.22;
  1224→      }
  1225→      // touch place button
  1226→      if (this.touchInput.placeBtn && this.placeCooldown <= 0) {
  1227→        this.tryPlace();
  1228→        this.placeCooldown = 0.22;
  1229→      }
  1230→    }
  1231→
  1232→    // block-break progress tracking + selection box: ONE raycast per frame
  1233→    if (this.inputEnabled) {
  1234→      const hit = this.player.raycast(REACH_DISTANCE);
  1235→      this._frameRaycast = hit;
  1236→      const breaking =
  1237→        (this.pointerLocked && this.mouseButtons.has(0)) ||
  1238→        this.touchInput.breakBtn;
  1239→      if (hit && breaking) {
  1240→        if (
  1241→          this.targetBlock &&
  1242→          this.targetBlock.x === hit.x &&
  1243→          this.targetBlock.y === hit.y &&
  1244→          this.targetBlock.z === hit.z
  1245→        ) {
  1246→          this.breakProgress = Math.min(1, this.breakProgress + dt * 2);
  1247→        } else {
  1248→          this.targetBlock = { x: hit.x, y: hit.y, z: hit.z };
  1249→          this.breakProgress = 0;
  1250→        }
  1251→      } else {
  1252→        this.targetBlock = null;
  1253→        this.breakProgress = 0;
  1254→      }
  1255→    } else {
  1256→      this._frameRaycast = null;
  1257→      this.targetBlock = null;
  1258→      this.breakProgress = 0;
  1259→    }
  1260→
  1261→    // player physics + survival
  1262→    this.player.update(dt);
  1263→
  1264→    // world streaming (budget scales modestly with render distance)
  1265→    const budget = Math.max(2, Math.floor(this.settings.renderDistance / 2));
  1266→    this.world.update(this.player.pos.x, this.player.pos.z, budget);
  1267→
  1268→    // weather + sky + particles
  1269→    this.updateWeather(dt);
  1270→    this.updateRain(dt);
  1271→    this.updateSky(dt);
  1272→    this.particles.update(dt);
  1273→    this.explosions.update(dt);
  1274→    this.updateSelectionBox();
  1275→
  1276→    // FOV kick (sprint) — smooth apply to camera
  1277→    const targetFov = this.settings.fov + this.player.fovKick;
  1278→    if (Math.abs(this.camera.fov - targetFov) > 0.05) {
  1279→      this.camera.fov += (targetFov - this.camera.fov) * Math.min(1, dt * 10);
  1280→      this.camera.updateProjectionMatrix();
  1281→    }
  1282→
  1283→    // render
  1284→    this.renderer.render(this.scene, this.camera);
  1285→
  1286→    // HUD updates (throttled)
  1287→    this.hudTimer += dt;
  1288→    if (this.hudTimer >= 0.15) {
  1289→      this.hudTimer = 0;
  1290→      this.emitHud();
  1291→    }
  1292→  };
  1293→
  1294→  private emitHud() {
